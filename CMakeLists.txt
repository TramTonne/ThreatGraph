cmake_minimum_required(VERSION 3.20)
project(ThreatGraph VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
find_package(nlohmann_json 3.11 QUIET)
if(NOT nlohmann_json_FOUND)
  FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
  FetchContent_MakeAvailable(json)
endif()
add_library(threatgraph engine/src/graph.cpp)
target_include_directories(threatgraph PUBLIC engine/include)
target_link_libraries(threatgraph PUBLIC nlohmann_json::nlohmann_json)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(threatgraph PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(threatgraph_engine engine/src/main.cpp)
target_link_libraries(threatgraph_engine PRIVATE threatgraph)
include(CTest)
if(BUILD_TESTING)
  add_executable(threatgraph_tests engine/tests/graph_tests.cpp)
  target_link_libraries(threatgraph_tests PRIVATE threatgraph)
  add_test(NAME engine_unit COMMAND threatgraph_tests)
  add_test(NAME engine_cli_example COMMAND threatgraph_engine ${CMAKE_CURRENT_SOURCE_DIR}/examples/office.json)
  add_test(NAME engine_cli_contract COMMAND ${CMAKE_COMMAND} -DENGINE=$<TARGET_FILE:threatgraph_engine> -DEXAMPLE=${CMAKE_CURRENT_SOURCE_DIR}/examples/office.json -P ${CMAKE_CURRENT_SOURCE_DIR}/engine/tests/cli_test.cmake)
endif()
